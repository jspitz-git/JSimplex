function _s3d_checked_basis_solve!(destination,ws,rhs,stop=nothing;transposed=false,
                               operation=transposed ? :btran : :ftran,prepare_update::Bool=false,
                               unit_row::Int=0)
    _timed_simplex(ws, transposed ? :btran : :ftran) do
        _pipeline_basis_solve!(destination,ws,rhs;transposed,operation,prepare_update,unit_row,kernel_mode=_s3d_weight_mode[])
    end
    _maybe_refine_basis_solve!(destination,ws,rhs,stop;transposed) || throw(_UnreliableBasisSolve())
    return destination
end
